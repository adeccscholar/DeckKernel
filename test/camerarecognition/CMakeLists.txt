cmake_minimum_required(VERSION 3.25)
get_filename_component(DECKKERNEL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ADECC_BOOTSTRAP_TOOLS "${DECKKERNEL_ROOT}/Cache/BootstrapTools.cmake")
if(NOT EXISTS "${ADECC_BOOTSTRAP_TOOLS}")
   message(FATAL_ERROR "Run cmake -P bootstrap/Bootstrap.cmake first")
endif()
include("${ADECC_BOOTSTRAP_TOOLS}")
set(CMAKE_CXX_COMPILER "${ADECC_BCC64X_EXECUTABLE}" CACHE FILEPATH "BCC64X compiler")
set(CMAKE_MAKE_PROGRAM "${ADECC_NINJA_EXECUTABLE}" CACHE FILEPATH "Ninja")
project(DeckKernelCameraRecognitionTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug" AND NOT CMAKE_BUILD_TYPE STREQUAL "Release")
   message(FATAL_ERROR "CMAKE_BUILD_TYPE must be Debug or Release")
endif()
set(ADECC_THIRDPARTY_CONFIGURATION "${CMAKE_BUILD_TYPE}")
set(CMAKE_INSTALL_PREFIX "${DECKKERNEL_ROOT}/apps/${CMAKE_BUILD_TYPE}" CACHE PATH "Apps" FORCE)
include("${DECKKERNEL_ROOT}/Cache/cmake/ThirdParty.cmake")
find_package(OpenCV 5.0 CONFIG REQUIRED COMPONENTS core imgproc imgcodecs geometry)
add_executable(deckkernel_camera_recognition_test main.cpp)
set_property(TARGET deckkernel_camera_recognition_test PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(deckkernel_camera_recognition_test PRIVATE CHRONO_WORK_AROUND CHRONO_MISSED_ZONEDB)
target_include_directories(deckkernel_camera_recognition_test PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(deckkernel_camera_recognition_test PRIVATE ${OpenCV_LIBS})
if(TARGET opencv_videoio)
   target_link_libraries(deckkernel_camera_recognition_test PRIVATE opencv_videoio)
   target_compile_definitions(deckkernel_camera_recognition_test PRIVATE DECKKERNEL_CAMERA_HAS_VIDEOIO=1)
   message(STATUS "CameraRecognitionTest: OpenCV videoio available")
else()
   message(STATUS "CameraRecognitionTest: no videoio; replay and self-test remain available")
endif()
# OCR uses the pinned BuildEngine packages only when explicitly enabled.
# The language model is separate runtime data, never downloaded by CMake.
option(DECKKERNEL_CAMERA_ENABLE_OCR "Enable pinned Tesseract OCR on stable card events" ON)
if(DECKKERNEL_CAMERA_ENABLE_OCR)
   find_package(Leptonica 1.87 CONFIG QUIET)
   find_package(Tesseract 5.5 CONFIG QUIET)
   if(TARGET Tesseract::libtesseract AND
      Tesseract_INCLUDE_DIRS AND
      EXISTS "${Tesseract_INCLUDE_DIRS}/tesseract/baseapi.h")
      target_include_directories(deckkernel_camera_recognition_test PRIVATE
         "${Tesseract_INCLUDE_DIRS}" "${Leptonica_INCLUDE_DIRS}")
      target_link_libraries(deckkernel_camera_recognition_test PRIVATE
         Tesseract::libtesseract)
      target_compile_definitions(deckkernel_camera_recognition_test PRIVATE
         DECKKERNEL_CAMERA_HAS_OCR=1)
      message(STATUS "CameraRecognitionTest: Tesseract OCR enabled; eng.traineddata still required at runtime")
   else()
      message(STATUS "CameraRecognitionTest: pinned Tesseract unavailable; OCR disabled")
   endif()
endif()
include("${DECKKERNEL_ROOT}/Cache/cmake/InstallPinnedRuntime.cmake")
adecc_stage_pinned_runtime(deckkernel_camera_recognition_test)
install(TARGETS deckkernel_camera_recognition_test RUNTIME DESTINATION .)
adecc_install_pinned_runtime(deckkernel_camera_recognition_test Runtime)
# Independent hardware activation probe. DirectShow's Windows SDK COM
# interfaces are used directly; OpenCV DLLs and its videoio wrapper are
# deliberately NOT linked into this executable.
add_executable(deckkernel_native_camera_probe native_camera.cpp)
set_property(TARGET deckkernel_native_camera_probe PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(deckkernel_native_camera_probe PRIVATE cxx_std_23)
target_compile_definitions(deckkernel_native_camera_probe PRIVATE
   UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
   CHRONO_WORK_AROUND CHRONO_MISSED_ZONEDB)
target_link_libraries(deckkernel_native_camera_probe PRIVATE ole32 oleaut32 user32)
adecc_stage_pinned_runtime(deckkernel_native_camera_probe)
install(TARGETS deckkernel_native_camera_probe RUNTIME DESTINATION .)
adecc_install_pinned_runtime(deckkernel_native_camera_probe Runtime)

enable_testing()
add_test(NAME CameraRecognitionStateMachine COMMAND deckkernel_camera_recognition_test --self-test)
