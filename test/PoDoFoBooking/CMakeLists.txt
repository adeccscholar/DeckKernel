cmake_minimum_required(VERSION 3.25)
project(adecc_podofo_booking LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Use your manually installed PoDoFo 1.1.1 stack. No DeckKernel bootstrap.
set(PODOFO_PACKAGE_DIR "" CACHE PATH "Directory containing podofo-config.cmake")
if(NOT PODOFO_PACKAGE_DIR)
   message(FATAL_ERROR "Set -DPODOFO_PACKAGE_DIR=... with locally installed PoDoFo 1.1.1")
endif()
if(NOT EXISTS "${PODOFO_PACKAGE_DIR}/podofo-config.cmake")
   message(FATAL_ERROR "PoDoFo config not found under PODOFO_PACKAGE_DIR")
endif()
find_package(podofo 1.1.1 CONFIG REQUIRED PATHS "${PODOFO_PACKAGE_DIR}" NO_DEFAULT_PATH)
if(NOT TARGET podofo::podofo)
   message(FATAL_ERROR "PoDoFo package does not export podofo::podofo")
endif()
add_library(adecc_pdf_booking_parser STATIC pdf_booking.cpp pdf_booking.h)
target_include_directories(adecc_pdf_booking_parser PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
add_executable(adecc_podofo_booking main.cpp)
target_link_libraries(adecc_podofo_booking PRIVATE adecc_pdf_booking_parser podofo::podofo)
enable_testing()
add_executable(adecc_podofo_booking_parser_test parser_test.cpp)
target_link_libraries(adecc_podofo_booking_parser_test PRIVATE adecc_pdf_booking_parser)
add_test(NAME PdfBookingSyntheticParser COMMAND adecc_podofo_booking_parser_test)
install(TARGETS adecc_podofo_booking RUNTIME DESTINATION . COMPONENT podofo_booking)
